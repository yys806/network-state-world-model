```python
from enum import IntEnum

import numpy as np


class MissingReason(IntEnum):
    NONE = 0
    NOT_APPLICABLE = 1
    NOT_COLLECTED = 2
    NO_HISTORY = 3
    SOURCE_ABSENT = 4
    SOURCE_ERROR = 5
    INVALID_VALUE = 6
    TIME_MISMATCH = 7
    SHAPE_MISMATCH = 8


def validate_masked_field(
    values: np.ndarray,
    valid_mask: np.ndarray,
    missing_reason: np.ndarray,
) -> None:
    values = np.asarray(values)
    valid_mask = np.asarray(valid_mask, dtype=bool)
    missing_reason = np.asarray(missing_reason)
    if values.shape != valid_mask.shape or values.shape != missing_reason.shape:
        raise ValueError("value, valid mask, and missing reason shapes differ")
    if not np.isfinite(values).all():
        raise ValueError("field contains NaN or Inf")
    if np.any(valid_mask & (missing_reason != MissingReason.NONE.value)):
        raise ValueError("valid element must use missing_reason=none")
    if np.any((~valid_mask) & (missing_reason == MissingReason.NONE.value)):
        raise ValueError("invalid element requires a non-none missing reason")
    if np.any((~valid_mask) & (values != 0)):
        raise ValueError("invalid element must use zero fill")


def validate_assignment_coo(
    assignment_coo: np.ndarray,
    capacities: tuple[int, int, int, int],
) -> None:
    coo = np.asarray(assignment_coo)
    if coo.dtype.kind not in "iu" or coo.ndim != 2 or coo.shape[1] != 4:
        raise ValueError("assignment_coo must be integer [record,4]")
    if len({tuple(row) for row in coo.tolist()}) != len(coo):
        raise ValueError("duplicate assignment")
    labels = ("time", "flow", "edge", "resource")
    for column, (label, capacity) in enumerate(zip(labels, capacities, strict=True)):
        if capacity < 0 or np.any(coo[:, column] < 0) or np.any(coo[:, column] >= capacity):
            raise ValueError(f"{label} index out of range")
```

实现`validate_edge_applicability()`，强制无线专属字段在wired边只能使用`not_applicable`；理论适用但未采集只能使用`not_collected/source_absent`：

```python
def validate_edge_applicability(
    field_name: str,
    edge_class: str,
    valid_mask: np.ndarray,
    missing_reason: np.ndarray,
) -> None:
    registry = {row["name"]: row for row in build_field_registry()}
    if field_name not in registry:
        raise ValueError(f"unknown field: {field_name}")
    if edge_class not in {"wireless", "wired"}:
        raise ValueError(f"unknown edge class: {edge_class}")
    valid_mask = np.asarray(valid_mask, dtype=bool)
    missing_reason = np.asarray(missing_reason)
    if valid_mask.shape != missing_reason.shape:
        raise ValueError("valid mask and missing reason shapes differ")
    applicable = edge_class in registry[field_name]["applicable_edge_classes"]
    if not applicable:
        if np.any(valid_mask) or np.any(missing_reason != MissingReason.NOT_APPLICABLE.value):
            raise ValueError("non-applicable field must be invalid with not_applicable reason")
        return
    if np.any((~valid_mask) & (missing_reason == MissingReason.NOT_APPLICABLE.value)):
        raise ValueError("applicable field cannot hide missing data as not_applicable")
```

def partition_trajectory_rows(
    rows: Iterable[Mapping[str, str]],

    selected: list[dict[str, str]] = []
    locked_metadata: list[dict[str, Any]] = []

def build_v3_fixture(root: Path, *, invalid_nonzero: bool = False) -> Path:
    root.mkdir(parents=True)
    fieldnames = ["trajectory_id", "seed", "split", "v3_seed_dir"]
    rows = [
        {"trajectory_id": "train_0", "seed": 0, "split": "train", "v3_seed_dir": "seed_000"},
        {"trajectory_id": "validation_0", "seed": 1, "split": "validation", "v3_seed_dir": "seed_001"},
        {"trajectory_id": "calibration_0", "seed": 2, "split": "calibration", "v3_seed_dir": "seed_002"},
        {"trajectory_id": "locked_0", "seed": 9, "split": "locked_test", "v3_seed_dir": "seed_009"},
    ]
    with (root / "trajectory_index.csv").open("w", encoding="utf-8", newline="") as handle:
        writer = csv.DictWriter(handle, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(rows)
    for row in rows[:3]:
        seed_dir = root / row["v3_seed_dir"]
        seed_dir.mkdir()
        values = np.zeros((2, 3, 18), np.float32)
        masks = np.zeros_like(values, dtype=bool)
        masks[..., [0, 1, 8, 11, 12]] = True
        values[..., 0] = 1.0
        values[..., 1] = 91.0
        if invalid_nonzero:
            values[..., 2] = 7.0
        np.savez_compressed(
            seed_dir / "trajectory_tensors.npz",
            information_edge_state=values,
            information_edge_feature_mask=masks,
        )
        tensor_path = seed_dir / "trajectory_tensors.npz"
        (seed_dir / "manifest.json").write_text(
            json.dumps({
                "schema_version": "PIJWM-DG-Contract-v3-dataset",
                "trajectory_id": row["trajectory_id"],
                "seed": row["seed"],
                "split": row["split"],
                "files": {
                    "trajectory_tensors.npz": {
                        "sha256": hashlib.sha256(tensor_path.read_bytes()).hexdigest(),
                        "size_bytes": tensor_path.stat().st_size,
                    }
                },
            }),
            encoding="utf-8",
        )
    locked_dir = root / "seed_009"
    locked_dir.mkdir()
    (locked_dir / "DO_NOT_OPEN").write_text("locked sentinel", encoding="utf-8")
