- [ ] **Step 1：实现安全规范化和稳定前缀映射**

```python
STABLE_PREFIXES = {
    "代码/": "code/",
    "文档/文献/": "literature/",
    "文档/组会/": "meeting/",
    "文档/知识库/": "记录/",
}


def resolve_repository_path(
    project_root: Path,
    raw_path: str,
    *,
    exact_mapping: Mapping[str, str] | None = None,
) -> Path:
    key = raw_path.replace("\\", "/")
    pure = PurePosixPath(key)
    if pure.is_absolute() or ".." in pure.parts:
        raise ValueError("path must be repository-relative")
    mapped = (exact_mapping or {}).get(key)
    if mapped is None:
        mapped = next(
            (new + key[len(old):] for old, new in STABLE_PREFIXES.items() if key.startswith(old)),
            key,
        )
    candidate = Path(project_root).joinpath(*PurePosixPath(mapped).parts)
    candidate.resolve(strict=False).relative_to(Path(project_root).resolve())
    return candidate
```

`文档/研究进展/**`只从精确迁移映射解析，避免把拆分类别误映射到单一目录。
